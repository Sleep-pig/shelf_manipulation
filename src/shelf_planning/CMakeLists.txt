cmake_minimum_required(VERSION 3.10)
project(shelf_planning)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/EvaluateCandidate.srv"
  DEPENDENCIES moveit_msgs trajectory_msgs geometry_msgs)
add_executable(evaluate src/evaluate.cpp)
ament_target_dependencies(evaluate moveit_ros_planning rclcpp)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(evaluate "${typesupport}")
install(TARGETS evaluate DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
