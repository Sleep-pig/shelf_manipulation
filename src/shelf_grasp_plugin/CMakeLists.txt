cmake_minimum_required(VERSION 3.10)
project(shelf_grasp_plugin)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
add_library(shelf_grasp SHARED src/grasp.cpp)
target_include_directories(shelf_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(shelf_grasp ${GAZEBO_LIBRARIES})
install(TARGETS shelf_grasp LIBRARY DESTINATION lib)
ament_package()
